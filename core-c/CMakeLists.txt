cmake_minimum_required(VERSION 3.16)
project(tm_c C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(tm_c STATIC src/tm_c.c)
target_include_directories(tm_c PUBLIC src)
target_compile_options(tm_c PRIVATE -Wall -Wextra)
add_executable(tm_c_test src/test_main.c)
target_link_libraries(tm_c_test PRIVATE tm_c m)
enable_testing()
add_test(NAME tm_c_test COMMAND tm_c_test)
